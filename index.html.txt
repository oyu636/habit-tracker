            <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" stroke-width="2" stroke="currentColor" class="w-7 h-7">

        <!-- JavaScriptでここにカードが生成されます -->
    </div>

    <script>
        // --- 1. タスクの初期データ定義 ---
        // name: タスク名
        // today: モックとしての「今日」の日付（破線で表示する用）
        const TASKS = [
            { id: 'face', name: '顔の運動7種', today: 6 },
            { id: 'gym', name: 'ジム1h(🏃‍♀️30m,他)', today: 6 },
            { id: 'thigh', name: '太もも', today: 6 },
            { id: 'plank', name: 'プランク1m', today: 6 },
            { id: 'back', name: '背中', today: 6 },
            { id: 'web', name: 'web関連制作1h', today: 6 }
        ];

        // 画像に合わせた初期状態のデータ（初回起動時のみ使用）
        const INITIAL_MOCK_DATA = {
            'face': [3, 5, 6],
            'gym': [],
            'thigh': [5, 6],
            'plank': [5, 6],
            'back': [6],
            'web': []
        };

        // --- 2. データ管理（localStorageの読み書き） ---
        let appData = {};

        function loadData() {
            const savedData = localStorage.getItem('habitTrackerData');
            if (savedData) {
                appData = JSON.parse(savedData);
            } else {
                // 初回アクセス時は画像のようなモックデータをセット
                appData = INITIAL_MOCK_DATA;
                saveData();
            }
        }

        function saveData() {
            localStorage.setItem('habitTrackerData', JSON.stringify(appData));
        }

        // --- 3. 達成日数の計算 ---
        // 選択された合計日数を計算します（連続日数ではなく全アクティブ日数）
        function calculateStreak(taskId, today) {
            const completedDays = appData[taskId] || [];
            return completedDays.length;
        }

        // --- 4. 日付をタップしたときの処理 ---
        function toggleDay(taskId, day, today) {
            if (!appData[taskId]) {
                appData[taskId] = [];
            }
            
            const index = appData[taskId].indexOf(day);
            if (index === -1) {
                // 未達成なら達成（配列に追加）
                appData[taskId].push(day);
            } else {
                // 達成済みなら未達成（配列から削除）
                appData[taskId].splice(index, 1);
            }
            
            saveData(); // localStorageに保存

            // 画面の更新 (DOMを直接操作して軽くする)
            const dayEl = document.getElementById(`day-${taskId}-${day}`);
            const streakEl = document.getElementById(`streak-${taskId}`);
            const fireIconEl = document.getElementById(`fire-${taskId}`);
            
            // 色の切り替え
            const isCompleted = index === -1; // さっき追加されたならtrue
            
            if (isCompleted) {
                dayEl.className = "flex items-center justify-center rounded text-xs font-bold w-full aspect-square cursor-pointer transition-colors bg-accent text-white";
            } else {
                // 未達成に戻す場合（今日のターゲットだった場合は破線に戻す）
                if (day === today) {
                    dayEl.className = "flex items-center justify-center rounded text-xs font-bold w-full aspect-square cursor-pointer transition-colors target-day text-daytext";
                } else {
                    dayEl.className = "flex items-center justify-center rounded text-xs font-bold w-full aspect-square cursor-pointer transition-colors bg-daybg text-daytext";
                }
            }

            // ストリークの更新
            const newStreak = calculateStreak(taskId, today);
            streakEl.textContent = newStreak;
            
            // 炎アイコンの色をストリークによって変える
            if (newStreak > 0) {
                fireIconEl.classList.remove('text-gray-600');
                fireIconEl.classList.add('text-accent');
            } else {
                fireIconEl.classList.remove('text-accent');
                fireIconEl.classList.add('text-gray-600');
            }
        }

        // --- 5. 画面の描画処理 ---
        function renderApp() {
            const container = document.getElementById('app-container');
            container.innerHTML = ''; // 一旦クリア

            TASKS.forEach(task => {
                const completedDays = appData[task.id] || [];
                const streakCount = calculateStreak(task.id, task.today);
                
                // カードの枠組み
                const card = document.createElement('div');
                card.className = "bg-cardbg rounded-3xl p-3 sm:p-4 flex flex-col justify-between shadow-lg";
                
                // カレンダーのグリッド生成 (7列 x 5行程度)
                let calendarHTML = `<div class="grid grid-cols-7 gap-1 sm:gap-1.5 mb-4">`;
                
                for (let day = 1; day <= 30; day++) {
                    let dayClasses = "flex items-center justify-center rounded text-xs font-bold w-full aspect-square cursor-pointer transition-colors ";
                    
                    if (completedDays.includes(day)) {
                        // 達成済み（赤色）
                        dayClasses += "bg-accent text-white";
                    } else if (day === task.today) {
                        // 今日（未達成の場合は破線）
                        dayClasses += "target-day text-daytext";
                    } else {
                        // 過去の未達成日（グレー）
                        dayClasses += "bg-daybg text-daytext";
                    }

                    // クリックイベントでtoggleDay関数を呼ぶ
                    calendarHTML += `<div id="day-${task.id}-${day}" class="${dayClasses}" onclick="toggleDay('${task.id}', ${day}, ${task.today})">${day}</div>`;
                }
                calendarHTML += `</div>`;

                // ストリーク（炎アイコンと数字）とタスク名部分
                const fireColorClass = streakCount > 0 ? 'text-accent' : 'text-gray-600';
                
                const footerHTML = `
                    <div class="flex flex-col items-center mt-auto">
                        <div class="flex items-center justify-center gap-1 mb-2">
                            <!-- 炎アイコン -->
                            <svg id="fire-${task.id}" xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="currentColor" class="w-5 h-5 sm:w-6 sm:h-6 ${fireColorClass}">
                                <path fill-rule="evenodd" d="M12..742 9.742 0 00-3.539 6.177A7.547 7.547 0 016.648 6.61a.75.75 0 00-1.152.082A9 9 0 1015.68 4.534a7.46 7.46 0 01-2.717-2.248zM15.75 14.25a3.75 3.75 0 11-7.313-1.172c.628.465 1.35.81 2.133 1a5.99 5.99 0 011.925-3.545 3.75 3.75 0 013.255 3.717z" clip-rule="evenodd" />
                            </svg>
                            <span id="streak-${task.id}" class="text-xl sm:text-2xl font-bold">${streakCount}</span>
                        </div>
                        <h3 class="text-sm sm:text-base font-bold text-center leading-tight truncate w-full">${task.name}</h3>
                    </div>
                `;

                card.innerHTML = calendarHTML + footerHTML;
                container.appendChild(card);
            });
        }

        // --- 6. アプリの起動 ---
        window.onload = () => {
            loadData();
            renderApp();
        };
    </script>
